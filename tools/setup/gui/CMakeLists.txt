cmake_minimum_required(VERSION 3.21)
project(RouterSetupPrototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Network)
qt_standard_project_setup()
qt_add_executable(router-setup WIN32 main.cpp Backend.h)
qt_add_qml_module(router-setup URI RouterSetup VERSION 1.0 QML_FILES qml/Main.qml qml/Theme.qml qml/Flow.js qml/RealPane.qml)
qt_add_resources(router-setup "photos" PREFIX "/images" BASE "../vendor/photos/v3.30" FILES "../vendor/photos/v3.30/tl-mr3020-3.jpg")
target_link_libraries(router-setup PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network)
if(MINGW)
 target_link_options(router-setup PRIVATE -static -static-libgcc -static-libstdc++)
endif()

# MSYS2 Qt static packages can refer to DLL import libraries for dependencies.
if(MINGW)
 get_property(imported DIRECTORY PROPERTY IMPORTED_TARGETS)
 foreach(t IN LISTS imported)
  foreach(prop IMPORTED_LOCATION IMPORTED_LOCATION_RELEASE IMPORTED_IMPLIB IMPORTED_IMPLIB_RELEASE INTERFACE_LINK_LIBRARIES)
   get_target_property(v ${t} ${prop})
   if(v)
    string(REPLACE ".dll.a" ".a" v "${v}")
    set_property(TARGET ${t} PROPERTY ${prop} "${v}")
   endif()
  endforeach()
 endforeach()
endif()

if(MINGW)
 qt_import_plugins(router-setup EXCLUDE_BY_TYPE imageformats INCLUDE Qt6::QJpegPlugin)
 set_property(TARGET Freetype::Freetype APPEND PROPERTY INTERFACE_LINK_LIBRARIES bz2)
 set_property(TARGET OpenSSL::Crypto APPEND PROPERTY INTERFACE_LINK_LIBRARIES crypt32)
endif()

if(MINGW)
 find_library(HARFBUZZ_STATIC NAMES libharfbuzz.a REQUIRED)
 set_property(TARGET harfbuzz::harfbuzz PROPERTY IMPORTED_LOCATION "${HARFBUZZ_STATIC}")
 set_property(TARGET harfbuzz::harfbuzz APPEND PROPERTY INTERFACE_LINK_LIBRARIES graphite2 rpcrt4 usp10)
endif()
