cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/wizard-version.txt" WIZARD_VERSION LIMIT_COUNT 1)
project(SchoolbellSetupWizard VERSION ${WIZARD_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
# Ask FindOpenSSL for static metadata before Qt discovers its dependencies.
if(MINGW)
 set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Network)
qt_standard_project_setup()
qt_add_executable(router-setup WIN32 main.cpp Backend.h FirmwareBackend.h FirmwareImages.h)
set_target_properties(router-setup PROPERTIES OUTPUT_NAME schoolbell-setup-wizard)
target_compile_definitions(router-setup PRIVATE SCHOOLBELL_WIZARD_VERSION="${PROJECT_VERSION}")
if(WIN32)
 configure_file(version.rc.in version.rc @ONLY)
 target_sources(router-setup PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
endif()
qt_add_qml_module(router-setup URI RouterSetup VERSION 1.0 QML_FILES qml/Main.qml qml/RouteChooser.qml qml/FirmwareRoute.qml qml/Theme.qml qml/RealPane.qml qml/PasswordField.qml qml/StyledComboBox.qml)
target_link_libraries(router-setup PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network)
if(MINGW)
 target_link_options(router-setup PRIVATE -static -static-libgcc -static-libstdc++)
endif()

# MSYS2 Qt static packages can refer to DLL import libraries for dependencies.
if(MINGW)
 get_property(imported DIRECTORY PROPERTY IMPORTED_TARGETS)
 foreach(t IN LISTS imported)
  foreach(prop IMPORTED_LOCATION IMPORTED_LOCATION_RELEASE IMPORTED_IMPLIB IMPORTED_IMPLIB_RELEASE INTERFACE_LINK_LIBRARIES)
   get_target_property(v ${t} ${prop})
   if(v)
    string(REPLACE ".dll.a" ".a" v "${v}")
    set_property(TARGET ${t} PROPERTY ${prop} "${v}")
   endif()
  endforeach()
 endforeach()
endif()

if(MINGW)
 qt_import_plugins(router-setup EXCLUDE_BY_TYPE imageformats INCLUDE Qt6::QJpegPlugin)
 set_property(TARGET Freetype::Freetype APPEND PROPERTY INTERFACE_LINK_LIBRARIES bz2)
 # Keep system dependencies after libcrypto for every consumer, including
 # Core-only tests (Qt 6.12 uses OpenSSL there too).
 set_property(TARGET OpenSSL::Crypto APPEND PROPERTY INTERFACE_LINK_LIBRARIES crypt32 ws2_32 bcrypt)
endif()

if(MINGW)
 find_library(HARFBUZZ_STATIC NAMES libharfbuzz.a REQUIRED)
 set_property(TARGET harfbuzz::harfbuzz PROPERTY IMPORTED_LOCATION "${HARFBUZZ_STATIC}")
 set_property(TARGET harfbuzz::harfbuzz APPEND PROPERTY INTERFACE_LINK_LIBRARIES graphite2 rpcrt4 usp10)
endif()
